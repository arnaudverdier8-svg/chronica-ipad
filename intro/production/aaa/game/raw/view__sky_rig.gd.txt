 121 func _fit_shadows() -> void:
 122 var cam: = get_viewport().get_camera_3d()
 123 if cam == None:
 124 return
 125 var vr: = get_viewport().get_visible_rect()
 126 var o: = cam.project_ray_origin(Vector2(vr.size.x * 0.5, 0))

 128 if dir.y > -0.01:
 129 sun.directional_shadow_max_distance = SHADOW_DIST

 131 var hit: = o + dir *(- o.y / dir.y)
