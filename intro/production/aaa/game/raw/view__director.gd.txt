 270 func _move(ev: Dictionary) -> void:
 271 var uid: = int(ev['uid'])

 273 if p == None:
 274 return
 275 var path: Array = ev['path']
 276 if path.is_empty():
 277 return
 278 var last: Vector2i = path[path.size() - 1]
 279 var seen: = False
 280 for c in path:
 281 if _seen(c):
 282 seen = True
 283 if bool(ev.get('instant', False)) or not seen:
 284 await _idle(p)
 285 p.position = _stand(uid, last)
 286 p.visible = _seen(last)
 287 return
 288 await _idle(p)
 289 p.visible = True
 290 var pts: Array =[]
 291 for c in path:
 292 pts.append(_stand(uid, c) if c == last else Vis.pos(g, c))
 293 var tw: = p.walk(pts, STEP_SECS)
 294 var u: = g.unit_by_id(uid)
 295 if u != None and u.owner == viewer and g.current == viewer:

 299 await _wait(AI_STAGGER)
