 275 func _synth_stream(kind: String) -> AudioStreamWAV:

 278 var rate: = 22050
 279 var secs: float = { 'coins': 0.7, 'step': 0.18, 'horn': 1.6, 'bell': 1.4 }[kind]
 280 var n: = int(rate * secs)
 281 var data: = PackedByteArray()

 283 var r: = RandomNumberGenerator.new()

 286 var t: = float(i) / rate
 287 var v: = 0
 288 match kind:
 289 'coins':

 292     var t0: = t - k * 0.11
 293     if t0 > 0:
 294     var f: = 2400 + k * 530
 295     v +=(sin(TAU * f * t0) * 0.6 + sin(TAU * f * 2.76 * t0) * 0.3) * exp(- t0 * 14)
 296     v *= 0.35
 297 'step':
 298     v =(r.randf() * 2 - 1) * exp(- t * 30) * 0.5

 306     v = v * env * 0.28

 308     for h in[1, 2, 2.4, 3, 4.2]:
 309     v += sin(TAU * 660 * h * t) * exp(- t *(3 + h)) / h
