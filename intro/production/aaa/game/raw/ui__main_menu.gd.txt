 133 func _layout() -> void:
 134 if not is_inside_tree() or _card == None:
 135 return
 136 var vr: = get_viewport_rect().size
 137 var ch: = _card.get_combined_minimum_size().y

 139 var tall: = _bar != None and vr.y >= ch + 230 and not Device.phone
 140 if _logo != None:
